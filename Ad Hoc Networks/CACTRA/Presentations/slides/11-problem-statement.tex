\begin{frame}{Problem Statement}
\begin{center}
\textbf{TCP traffic unfairness comes from both DCF channel access and TCP traffic-rate control.}
\end{center}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{DCF channel-access behavior}
\begin{itemize}
    \item BEB does not necessarily ensure per-station fairness.
    \item A congestion-based CW may disadvantage some stations.
\end{itemize}

\column{0.48\textwidth}
\textbf{TCP traffic-rate behavior}
\begin{itemize}
    \item An excessively large $cwnd$ creates too many packets.
    \item Some flows obtain much more bandwidth than others.
\end{itemize}
\end{columns}
